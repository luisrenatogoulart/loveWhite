  <script>

    /* TIMER */
    function atualizarTimerEl(el){
      const inicio=new Date('2025-08-22T00:00:00');
      const agora=new Date();
      const diff=agora-inicio;
      const d=Math.floor(diff/(1000*60*60*24));
      const h=Math.floor((diff/(1000*60*60))%24);
      const m=Math.floor((diff/(1000*60))%60);
      const s=Math.floor((diff/1000)%60);
      el.textContent=`${d} dias · ${h}h ${m}m ${s}s`;
    }
    const timer=document.getElementById("timer");
    setInterval(()=>atualizarTimerEl(timer),1000);
    atualizarTimerEl(timer);

    /* PARTÍCULAS */
    function criarParticulas(){
      for(let i=0;i<12;i++){
        const h=document.createElement("div");
        h.className="heart";h.textContent="💖";
        h.style.left=Math.random()*100+"vw";
        h.style.top="100vh";
        h.style.transition="transform 6s linear,opacity 6s";
        document.body.appendChild(h);
        setTimeout(()=>{h.style.transform="translateY(-120vh)";h.style.opacity="0"},100);
        setTimeout(()=>h.remove(),6500);
      }
    }
    setInterval(criarParticulas,2000);

    /* CARROSSEL */
    const slides=[...document.querySelectorAll(".slide")];
    let current=0;

    function layout(){
      slides.forEach((s,i)=>{
        const pos=i-current;
        s.style.setProperty("--pos",pos);
        s.dataset.pos=pos;
        s.classList.toggle("hidden",Math.abs(pos)>3);
      });
    }
    layout();

    document.getElementById("prev").onclick=()=>{ current=(current-1+slides.length)%slides.length; layout(); }
    document.getElementById("next").onclick=()=>{ current=(current+1)%slides.length; layout(); }

    /* MÚSICA */
    const audio=document.getElementById("audio");
    const musicBtn=document.getElementById("musicBtn");
    let tocando=true;

    musicBtn.onclick=()=>{
      if(tocando){
        audio.pause();
        musicBtn.textContent="🔇";
        musicBtn.classList.add("paused");
      }else{
        audio.play();
        musicBtn.textContent="🎵";
        musicBtn.classList.remove("paused");
      }
      tocando=!tocando;
    };

    document.getElementById("startBtn").onclick=()=>{
      audio.play().catch(()=>{});
      document.getElementById("splash").style.opacity="0";
      setTimeout(()=>document.getElementById("splash").remove(),600);
    };

  </script>
